7 # to start, we're going to use a very simple definition of `include`
8 # that doesn't require any function (except what we can get from the `Core` top-module)
9 # start this big so that we don't have to resize before we have defined how to grow an array
10 const _included_files = Array{Tuple{Module,String},1}(Core.undef, 400)
11 setfield!(_included_files, :size, (1,))
12 function include(mod::Module, path::String)
13 len = getfield(_included_files.size, 1)
14 memlen = _included_files.ref.mem.length
15 lenp1 = Core.add_int(len, 1)
16 if len === memlen # by the time this is true we hopefully will have defined _growend!
17 _growend!(_included_files, UInt(1))
18 else
19 setfield!(_included_files, :size, (lenp1,))
20 end
21 Core.memoryrefset!(Core.memoryref(_included_files.ref, lenp1), (mod, ccall(:jl_prepend_cwd, Any, (Any,), path)), :not_atomic, true)
22 Core.println(path)
23 ccall(:jl_uv_flush, Nothing, (Ptr{Nothing},), Core.io_pointer(Core.stdout))
24 Core.include(mod, path)
25 end

577 # triggers printing the report and (optionally) saving a heap snapshot after a SIGINFO/SIGUSR1 profile request
578 # Needs to be in Base because Profile is no longer loaded on boot
579 function profile_printing_listener(cond::Base.AsyncCondition)
580 profile = nothing
581 try
582 while _trywait(cond)
583 profile = @something(profile, require_stdlib(PkgId(UUID("9abbd945-dff8-562f-b5e8-e1ebf5ef1b79"), "Profile")))::Module
584 invokelatest(profile.peek_report[])
585 if Base.get_bool_env("JULIA_PROFILE_PEEK_HEAP_SNAPSHOT", false) === true
586 println(stderr, "Saving heap snapshot...")
587 fname = invokelatest(profile.take_heap_snapshot)
588 println(stderr, "Heap snapshot saved to `$(fname)`")
589 end
590 end
591 catch ex
592 if !isa(ex, InterruptException)
593 @error "Profile printing listener crashed" exception=ex,catch_backtrace()
594 end
595 end
596 nothing
597 end
