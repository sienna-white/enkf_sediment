585 #l_comparison = length((unwrap_unionall(comparison)::DataType).parameters)
586 #spec_len = max(spec_len, l_comparison)

592 newsig = limit_type_size(sig, comparison, hardlimit ? comparison : mi.specTypes, InferenceParams(interp).tuple_complexity_limit_depth, spec_len)

596 # on this call (to ensure convergence), so that we don't cache this result

598 add_remark!(interp, sv, RECURSION_UNUSED_MSG)
599 # if we don't (typically) actually care about this result,
600 # don't bother trying to examine some complex abstract signature
601 # since it's very unlikely that we'll try to inline this,
602 # or want make an invoke edge to its calling convention return type.
603 # (non-typically, this means that we lose the ability to detect a guaranteed StackOverflow in some cases)
604 return MethodCallResult(Any, Any, true, true, nothing, Effects())
605 end
606 add_remark!(interp, sv, washardlimit ? RECURSION_MSG_HARDLIMIT : RECURSION_MSG)
607 # TODO (#48913) implement a proper recursion handling for irinterp:
608 # This works just because currently the `:terminate` condition usually means this is unreachable here
609 # for irinterp because there are not unresolved cycles, but it's not a good solution.
610 # We should revisit this once we have a better story for handling cycles in irinterp.
611 if isa(sv, InferenceState)
612 # since the hardlimit is against the edge to the parent frame,
613 # we should try to poison the whole edge, not just the topmost frame
614 parentframe = frame_parent(topmost)
615 while !isa(parentframe, InferenceState)
616 # attempt to find a parent frame that can handle this LimitedAccuracy result correctly
617 # so we don't try to cache this incomplete intermediate result
618 parentframe === nothing && break
619 parentframe = frame_parent(parentframe)
620 end
621 if isa(parentframe, InferenceState)
622 poison_callstack!(sv, parentframe)
623 elseif isa(topmost, InferenceState)
624 poison_callstack!(sv, topmost)

627 # n.b. this heuristic depends on the non-local state, so we must record the limit later
628 sig = newsig

630 edgelimited = true
631 end
632 end

634 # if sig changed, may need to recompute the sparams environment
635 if isa(method.sig, UnionAll) && isempty(sparams)
636 recomputed = ccall(:jl_type_intersection_with_env, Any, (Any, Any), sig, method.sig)::SimpleVector
637 #@assert recomputed[1] !== Bottom
638 # We must not use `sig` here, since that may re-introduce structural complexity that
639 # our limiting heuristic sought to eliminate. The alternative would be to not increment depth over covariant contexts,
640 # but we prefer to permit inference of tuple-destructuring, so we don't do that right now
641 # For example, with a signature such as `Tuple{T, Ref{T}} where {T <: S}`
642 # we might want to limit this to `Tuple{S, Ref}`, while type-intersection can instead give us back the original type
643 # (which moves `S` back up to a lower comparison depth)
644 # Optionally, we could try to drive this to a fixed point, but I think this is getting too complex,
645 # and this would only cause more questions and more problems
646 # (the following is only an example, most of the statements are probable in the wrong order):
647 # newsig = sig
648 # seen = IdSet()
649 # while !(newsig in seen)
650 # push!(seen, newsig)
651 # lsig = length((unwrap_unionall(sig)::DataType).parameters)
652 # newsig = limit_type_size(newsig, sig, sv.linfo.specTypes, InferenceParams(interp).tuple_complexity_limit_depth, lsig)
653 # recomputed = ccall(:jl_type_intersection_with_env, Any, (Any, Any), newsig, method.sig)::SimpleVector
654 # newsig = recomputed[2]
655 # end
656 # sig = ?
657 sparams = recomputed[2]::SimpleVector

660 6 (600.00%)
1 (16.67%) samples spent calling typeinf_edge
(; rt, exct, edge, effects, volatile_inf_result) = typeinf_edge(interp, method, sig, sparams, sv)
661
662 if edge === nothing
663 edgecycle = edgelimited = true
664 end
665
666 # we look for the termination effect override here as well, since the :terminates effect
667 # may have been tainted due to recursion at this point even if it's overridden
668 if is_effect_overridden(sv, :terminates_globally)
669 # this frame is known to terminate
670 effects = Effects(effects, terminates=true)
671 elseif is_effect_overridden(method, :terminates_globally)
672 # this edge is known to terminate
673 effects = Effects(effects; terminates=true)
674 elseif edgecycle
675 # Some sort of recursion was detected.
676 if edge !== nothing && !edgelimited && !is_edge_recursed(edge, sv)
677 # no `MethodInstance` cycles -- don't taint :terminate
678 else
679 # we cannot guarantee that the call will terminate
680 effects = Effects(effects; terminates=false)
681 end
682 end
683
684 return MethodCallResult(rt, exct, edgecycle, edgelimited, edge, effects, volatile_inf_result)

1019 function const_prop_entry_heuristic(interp::AbstractInterpreter, result::MethodCallResult,
1020 si::StmtInfo, sv::AbsIntState, force::Bool)
1021 if result.rt isa LimitedAccuracy
1022 # optimizations like inlining are disabled for limited frames,
1023 # thus there won't be much benefit in constant-prop' here
1024 # N.B. don't allow forced constprop' for safety (xref #52763)
1025 add_remark!(interp, sv, "[constprop] Disabled by entry heuristic (limited accuracy)")
1026 return false
1027 elseif force
1028 return true
1029 elseif call_result_unused(si) && result.edgecycle
1030 add_remark!(interp, sv, "[constprop] Disabled by entry heuristic (edgecycle with unused result)")
1031 return false
1032 end
1033 # check if this return type is improvable (i.e. whether it's possible that with more
1034 # information, we might get a more precise type)
1035 rt = result.rt
1036 if isa(rt, Type)
1037 # could always be improved to `Const`, `PartialStruct` or just a more precise type,
1038 # unless we're already at `Bottom`
1039 if rt === Bottom
1040 add_remark!(interp, sv, "[constprop] Disabled by entry heuristic (erroneous result)")
1041 return false
1042 end
1043 return true
1044 elseif isa(rt, PartialStruct) || isa(rt, InterConditional) || isa(rt, InterMustAlias)
1045 # could be improved to `Const` or a more precise wrapper
1046 return true
1047 elseif isa(rt, Const)
1048 if is_nothrow(result.effects)
1049 add_remark!(interp, sv, "[constprop] Disabled by entry heuristic (nothrow const)")
1050 return false
1051 end
1052 # Could still be improved to Bottom (or at least could see the effects improved)
1053 return true
1054 end
1055 add_remark!(interp, sv, "[constprop] Disabled by entry heuristic (unimprovable result)")
1056 return false
1057 end
